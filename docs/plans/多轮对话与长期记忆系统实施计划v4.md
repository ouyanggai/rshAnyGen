# 多轮对话与长期记忆系统实施计划

> **核心目标**: 实现基于主流方案的多轮对话、长期记忆系统

## 实施概览

本计划基于已完成的认证系统，专注实现以下核心能力：

1. **Token-aware 三级上下文管理** - 智能Token预算分配

## Phase 1: Token-aware 上下文管理 (Week 1)

### 目标

### 核心概念

**三级上下文架构**:
```
┌─────────────────────────────────────┐
│ 长期记忆层 (~200 tokens)             │
│ - 用户画像                           │
│ - 技能标签                           │
└─────────────────────────────────────┘
           ↓
┌─────────────────────────────────────┐
│ 短期摘要层 (~500 tokens)             │
│ - 主题分段摘要                       │
│ - 关键决策点                         │
└─────────────────────────────────────┘
           ↓
┌─────────────────────────────────────┐
│ 工作记忆层 (~2048 tokens)            │
│ - 最近3-10轮对话                     │
│ - Token动态裁剪                      │
└─────────────────────────────────────┘
           ↓
┌─────────────────────────────────────┐
│ 当前消息 (~100 tokens)               │
└─────────────────────────────────────┘
```

class TokenCounter:
    """增强的Token计数器"""
    
    async def estimate_response_tokens(
        self,
        context: List[Dict],
        model: str
    ) -> int:
        """估算模型响应会消耗的tokens"""
        # 基于历史数据估算
        # 通常响应长度 = 输入长度 * 0.3-0.5
        input_tokens = self.count_messages(context)
        return int(input_tokens * 0.4)
    
    def get_token_budget(
        self,
        model: str,
        max_output_tokens: int = 2000
    ) -> Dict[str, int]:
        """获取模型的Token预算分配"""
        model_limits = {
            "qwen-max": 30000,
            "qwen-plus": 30000,
            "gpt-4": 8192,
            "gpt-3.5-turbo": 16384
        }
        
        context_limit = model_limits.get(model, 8192)
        available = context_limit - max_output_tokens - 100  # 保留缓冲
        
        return {
            "total": context_limit,
            "available_for_context": available,
            "long_term": int(available * 0.1),      # 10%
            "short_term": int(available * 0.2),     # 20%
            "working": int(available * 0.7),        # 70%
        }
```

class ContextBuilder:
    async def build_context(
        self,
        session_id: str,
        user_id: str,
        current_message: str,
        model: str = "qwen-max"
    ) -> List[Dict]:
        """构建Token-aware上下文"""
        
        # 1. 获取Token预算
        budget = self.token_counter.get_token_budget(model)
        
        # 2. 构建各层上下文
        context = []
        used_tokens = 0
        
        # 长期记忆
        long_term = await self._build_long_term_memory(
            user_id, 
            current_message,
            max_tokens=budget["long_term"]
        )
        if long_term:
            context.append(long_term)
            used_tokens += self.token_counter.count_message(long_term)
        
        # 短期摘要
        summary = await self._build_summary_context(
            session_id,
            max_tokens=budget["short_term"]
        )
        if summary:
            context.extend(summary)
            used_tokens += sum(
                self.token_counter.count_message(m) for m in summary
            )
        
        # 工作记忆 (动态调整)
        remaining = budget["working"] - used_tokens
        working = await self._build_working_memory(
            session_id,
            max_tokens=max(remaining, 1024)  # 至少保留1K
        )
        context.extend(working)
        
        # 当前消息
        context.append({"role": "user", "content": current_message})
        
        return context
```

    self,
    user_id: str,
    current_message: str,
    max_tokens: int = 200
) -> Optional[Dict]:
    """构建相关的长期记忆"""
    
    # 1. 获取用户画像
    user_profile = await self.redis.hgetall(f"user:{user_id}")
    
    # 2. 检索相关实体
    relevant_entities = await self._retrieve_relevant_entities(
        user_id, 
        current_message,
        limit=3
    )
    
    # 3. 构建系统提示
    content_parts = []
    
    if user_profile:
        content_parts.append(
            f"用户: {user_profile.get('nickname', '用户')}"
        )
    
    if relevant_entities:
        entity_text = ", ".join([
            f"{e['name']}({e['type']})" 
            for e in relevant_entities
        ])
        content_parts.append(f"相关信息: {entity_text}")
    
    if not content_parts:
        return None
    
    content = "【背景信息】\n" + "\n".join(content_parts)
    
    # 4. Token限制检查
    if self.token_counter.count_text(content) > max_tokens:
        content = content[:max_tokens * 4]  # 粗略裁剪
    
    return {
        "role": "system",
        "content": content
    }

### Task 2.1: 主题切换检测

**文件**: `apps/orchestrator/services/topic_detector.py`

```python
class TopicDetector:
    """主题切换检测器"""
    
    def __init__(self, embedding_client, threshold: float = 0.7):
        self.embedding_client = embedding_client
        self.threshold = threshold
    
    async def detect_topic_change(
        self,
        messages: List[Dict],
        window_size: int = 4
    ) -> bool:
        """检测主题是否切换
        
        Args:
            messages: 消息列表
            window_size: 检测窗口大小
        
        Returns:
            True if topic changed
        """
        user_messages = [
            m for m in messages[-window_size:]
            if m.get("role") == "user"
        ]
        
        if len(user_messages) < 2:
            return False
        
        # 比较最近两条消息的相似度
        recent = user_messages[-2:]
        embeddings = await self.embedding_client.embed_batch([
            m["content"] for m in recent
        ])
        
        similarity = self._cosine_similarity(
            embeddings[0], 
            embeddings[1]
        )
        
        return similarity < self.threshold
    
    def _cosine_similarity(self, v1: List[float], v2: List[float]) -> float:
        """计算余弦相似度"""
        import numpy as np
        a = np.array(v1)
        b = np.array(v2)
        return np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b))
    
    async def segment_by_topic(
        self,
        messages: List[Dict],
        min_segment_size: int = 6
    ) -> List[Dict]:
        """按主题分段
        
        Returns:
            List of segments: [
                {
                    "start_idx": 0,
                    "end_idx": 5,
                    "messages": [...],
                    "topic": None
                }
            ]
        """
        segments = []
        current_start = 0
        
        for i in range(min_segment_size, len(messages), 2):
            if await self.detect_topic_change(messages[:i+1]):
                segments.append({
                    "start_idx": current_start,
                    "end_idx": i - 1,
                    "messages": messages[current_start:i],
                    "topic": None
                })
                current_start = i
        
        # 最后一段
        if current_start < len(messages):
            segments.append({
                "start_idx": current_start,
                "end_idx": len(messages) - 1,
                "messages": messages[current_start:],
                "topic": None
            })
        
        return segments
```

```python
class SummaryGenerator:
    """智能摘要生成器"""
    
    def __init__(self, llm_client, embedding_client):
        self.llm_client = llm_client
        self.topic_detector = TopicDetector(embedding_client)
        self.redis = RedisOperations()
    
    async def should_generate_summary(
        self,
        session_id: str,
        trigger_token_count: int = 8192
    ) -> bool:
        """判断是否需要生成摘要"""
        messages = await self.redis.lrange_json(
            f"session:messages:{session_id}",
            0,
            -1
        )
        
        token_counter = get_token_counter()
        total_tokens = token_counter.count_messages(messages)
        
        return total_tokens > trigger_token_count
    
    async def generate_summary(
        self,
        session_id: str
    ) -> List[Dict]:
        """生成分段摘要"""
        
        # 1. 获取所有消息
        messages = await self.redis.lrange_json(
            f"session:messages:{session_id}",
            0,
            -1
        )
        
        # 2. 按主题分段
        segments = await self.topic_detector.segment_by_topic(messages)
        
        logger.info(
            f"Session {session_id} segmented into {len(segments)} topics"
        )
        
        # 3. 为每段生成摘要
        summaries = []
        for segment in segments:
            summary = await self._summarize_segment(segment["messages"])
            
            summaries.append({
                "topic": summary["topic"],
                "summary": summary["content"],
                "message_range": [
                    segment["start_idx"],
                    segment["end_idx"]
                ],
                "created_at": datetime.now().isoformat()
            })
        
        # 4. 保存摘要
        await self.redis.delete(f"session:summaries:{session_id}")
        for summary in summaries:
            await self.redis.rpush_json(
                f"session:summaries:{session_id}",
                summary
            )
        
        return summaries
    
    async def _summarize_segment(
        self,
        messages: List[Dict]
    ) -> Dict:
        """为单个主题段生成摘要"""
        
        # 格式化对话
        conversation = []
        for msg in messages:
            role = "用户" if msg["role"] == "user" else "AI"
            conversation.append(f"{role}: {msg['content']}")
        
        conv_text = "\n".join(conversation)
        
        prompt = f"""分析以下对话，生成简洁摘要。

对话内容:
{conv_text}

要求:
1. 用3-5个字概括主题
2. 用1-2句话总结关键信息
3. 保留重要决策和结论
4. 忽略闲聊和重复内容

请以JSON格式输出:
{{
  "topic": "主题",
  "content": "摘要内容"
}}
"""
        
        response = await self.llm_client.complete(
            prompt,
            max_tokens=200,
            temperature=0.3
        )
        
        try:
            import json
            result = json.loads(response)
            return result
        except:
            return {
                "topic": "对话片段",
                "content": "摘要生成失败"
            }
```

class BackgroundTaskProcessor:

    def __init__(self):

        """启动后台任务"""

            self._process_summaries(),
            self._process_memory_extraction()

```python
class EntityExtractor:
    """实体提取器 (规则 + LLM 混合)"""
    
    # 提取规则
    PATTERNS = {
        "person": [
            r"(?:我|他|她)(?:叫|名字是)\s*(\w{2,10})",
            r"(?:同事|朋友|老师|领导)\s*(\w{2,10})",
        ],
        "project": [
            r"(?:项目|系统|产品)(?:叫|名为)\s*(\w{2,20})",
        ],
        "location": [
            r"(?:在|位于)\s*([\u4e00-\u9fa5]{2,10})",
        ]
    }
    
    def __init__(self, llm_client=None):
        self.llm_client = llm_client
        self.redis = RedisOperations()
    
    async def extract_entities(
        self,
        user_message: str,
        ai_response: str,
        user_id: str,
        session_id: str
    ) -> List[Entity]:
        """提取实体"""
        
        entities = []
        
        # 1. 规则提取 (快速、准确)
        rule_entities = self._extract_by_rules(user_message)
        entities.extend(rule_entities)
        
        # 2. LLM提取 (补充、深度)
        if self.llm_client and self._should_use_llm(user_message):
            llm_entities = await self._extract_by_llm(
                user_message,
                ai_response
            )
            entities.extend(llm_entities)
        
        # 3. 去重
        entities = self._deduplicate(entities)
        
        # 4. 保存
        for entity in entities:
            await self._save_entity(entity, user_id, session_id)
        
        return entities
    
    def _extract_by_rules(self, text: str) -> List[Entity]:
        """规则提取"""
        entities = []
        
        for entity_type, patterns in self.PATTERNS.items():
            for pattern in patterns:
                matches = re.findall(pattern, text)
                for match in matches:
                    entities.append(Entity(
                        type=entity_type,
                        name=match.strip(),
                        confidence=0.85
                    ))
        
        return entities
    
    async def _extract_by_llm(
        self,
        user_msg: str,
        ai_msg: str
    ) -> List[Entity]:
        """LLM提取"""
        
        prompt = f"""从对话中提取关键实体。

用户: {user_msg}
AI: {ai_msg}

提取类型: person(人名), project(项目), location(地点), concept(概念)

JSON格式输出:
{{
  "entities": [
    {{"type": "person", "name": "张三", "attributes": {{"role": "同事"}}, "confidence": 0.9}}
  ]
}}
"""
        
        try:
            response = await self.llm_client.complete(
                prompt,
                max_tokens=300,
                temperature=0.2
            )
            
            import json
            result = json.loads(response)
            return [Entity(**e) for e in result.get("entities", [])]
        
        except Exception as e:
            logger.error(f"LLM entity extraction failed: {e}")
            return []
    
    async def _save_entity(
        self,
        entity: Entity,
        user_id: str,
        session_id: str
    ):
        """保存实体到Redis"""
        
        # 检查是否已存在
        existing_id = await self._find_existing(entity, user_id)
        
        if existing_id:
            # 更新访问计数
            await self.redis.hincrby(
                f"entity:{existing_id}",
                "mention_count",
                1
            )
        else:
            # 创建新实体
            entity_id = f"entity-{entity.type}-{uuid.uuid4().hex[:8]}"
            
            entity_data = {
                "type": entity.type,
                "name": entity.name,
                "attributes": json.dumps(entity.attributes),
                "first_seen": datetime.now().isoformat(),
                "last_seen": datetime.now().isoformat(),
                "mention_count": 1,
                "session_id": session_id
            }
            
            await self.redis.hset(
                f"entity:{entity_id}",
                mapping=entity_data
            )
            
            # 加入用户实体索引
            await self.redis.zadd(
                f"user:entities:{entity.type}:{user_id}",
                {entity_id: int(time.time())}
            )
```

### Task 4.1: 记忆去重

```python
class MemoryService:
    """记忆管理服务"""
    
    async def save_memory_with_dedup(
        self,
        user_id: str,
        content: str,
        importance: float,
        session_id: str
    ):
        """保存记忆(带去重)"""
        
        # 1. 生成embedding
        embedding = await self.embedding_client.embed(content)
        
        # 2. 检查重复
        existing_id = await self.check_duplicate(
            user_id,
            embedding,
            content
        )
        
        if existing_id:
            # 更新现有记忆
            await self._update_memory_access(existing_id)
            logger.info(f"Memory deduplicated: {existing_id}")
            return
        
        # 3. 保存新记忆
        await self._save_new_memory(
            user_id,
            content,
            importance,
            embedding,
            session_id
        )
    
    async def check_duplicate(
        self,
        user_id: str,
        embedding: List[float],
        content: str,
        similarity_threshold: float = 0.92
    ) -> Optional[int]:
        """检查记忆是否重复"""
        
        # 搜索相似记忆 (最近30天)
        thirty_days_ago = int(time.time()) - (30 * 86400)
        
        results = self.collection.search(
            data=[embedding],
            anns_field="vector",
            param={"metric_type": "COSINE", "params": {"ef": 64}},
            limit=5,
            expr=f"user_id == '{user_id}' and created_at >= {thirty_days_ago}",
            output_fields=["id", "content", "importance"]
        )
        
        if not results or not results[0]:
            return None
        
        for hit in results[0]:
            similarity = 1 - hit.distance
            
            # 高相似度 -> 重复
            if similarity > similarity_threshold:
                return hit.id
            
            # 中等相似度 -> 合并
            elif similarity > 0.80:
                await self._merge_memories(hit.id, content)
                return hit.id
        
        return None
```

### Task 4.2: 时间衰减

**文件**: `apps/orchestrator/services/memory_scorer.py`

```python
class MemoryScorer:
    """记忆评分器(时间衰减)"""
    
    def __init__(self, half_life_days: int = 30):
        self.half_life_days = half_life_days
    
    def calculate_current_importance(
        self,
        base_importance: float,
        created_at: datetime,
        access_count: int = 0,
        last_accessed: datetime = None
    ) -> float:
        """计算当前重要性
        
        公式:
        current = base * time_decay + access_boost + recent_boost
        """
        import math
        
        # 1. 时间衰减 (指数)
        days_old = (datetime.now() - created_at).days
        time_decay = math.exp(-days_old / self.half_life_days)
        
        # 2. 访问频率加成 (最多+0.3)
        access_boost = min(access_count * 0.05, 0.3)
        
        # 3. 最近访问加成 (7天内)
        recent_boost = 0
        if last_accessed:
            days_since = (datetime.now() - last_accessed).days
            if days_since < 7:
                recent_boost = 0.2 * (1 - days_since / 7)
        
        # 4. 综合评分
        score = min(
            base_importance * time_decay + access_boost + recent_boost,
            1.0
        )
        
        return score
    
    def should_archive(
        self,
        current_importance: float,
        threshold: float = 0.1
    ) -> bool:
        """判断是否应该归档"""
        return current_importance < threshold
```

# 在 BackgroundTaskProcessor 类中添加

async def _process_memory_decay(self):
    """处理记忆衰减(每天一次)"""
    
    while self.running:
        try:
            logger.info("Starting memory decay processing")
            
            scorer = MemoryScorer(half_life_days=30)
            memory_service = MemoryService()
            
            # 获取所有用户
            user_keys = await self.redis.keys("user:*")
            user_ids = [
                k.split(":")[1] 
                for k in user_keys 
                if k.count(":") == 1
            ]
            
            for user_id in user_ids:
                # 更新用户所有记忆的评分
                await self._update_user_memory_scores(
                    user_id,
                    scorer,
                    memory_service
                )
            
            logger.info("Memory decay processing completed")
            
            # 每天执行一次
            await asyncio.sleep(86400)
        
        except Exception as e:
            logger.error(f"Memory decay error: {e}")
            await asyncio.sleep(3600)

async def _update_user_memory_scores(
    self,
    user_id: str,
    scorer: MemoryScorer,
    memory_service: MemoryService
):
    """更新用户记忆评分"""
    
    # 获取用户所有记忆
    results = memory_service.collection.query(
        expr=f"user_id == '{user_id}'",
        output_fields=[
            "id", "importance", "created_at",
            "access_count", "last_accessed"
        ]
    )
    
    for memory in results:
        # 计算当前重要性
        current_score = scorer.calculate_current_importance(
            base_importance=memory["importance"],
            created_at=datetime.fromisoformat(memory["created_at"]),
            access_count=memory.get("access_count", 0),
            last_accessed=datetime.fromisoformat(memory["last_accessed"])
                if memory.get("last_accessed") else None
        )
        
        # 判断是否归档
        if scorer.should_archive(current_score):
            await memory_service.archive_memory(memory["id"])
        else:
            # 更新评分
            await memory_service.update_score(
                memory["id"],
                current_score
            )

## 常见问题

### Q1: Token预算如何动态调整?

A: 根据模型上限和预期输出长度计算:
```python
available = model_limit - max_output - buffer
long_term = available * 0.1
short_term = available * 0.2
working = available * 0.7
```

### Q2: 主题切换检测不准确怎么办?

A: 
1. 调整相似度阈值 (默认0.7)
2. 增加检测窗口大小 (默认4)
3. 使用更好的embedding模型

### Q3: 记忆去重太激进怎么办?

A:
1. 提高相似度阈值 (从0.92到0.95)
2. 缩短检查时间范围 (从30天到7天)
3. 只去重完全相同的记忆

### Q4: 时间衰减太快怎么办?

A: 调整半衰期参数:
```python
scorer = MemoryScorer(half_life_days=60)  # 从30改为60
```
